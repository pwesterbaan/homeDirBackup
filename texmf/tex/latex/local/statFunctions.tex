% approximate probabilities calculated using the Normal distribution
%https://tex.stackexchange.com/a/759031/190910
\xintdeffloatfunc snapprox(x,h) = 1 - 0.5 * h * exp(
  -0.5*x^2 -1.26551223
  +1.00002368*h    +0.37409196*h^2
  +0.09678418*h^3 -0.18628806*h^4
  +0.27886807*h^5 -1.13520398*h^6
  +1.48851587*h^7 -0.82215223*h^8
  +0.17087277*h^9);

\xintNewFloatExpr{\sn}[1]{ snapprox(#1, 1/(1+0.25*sqrt(2)*#1))}

\ExplSyntaxOn
\NewDocumentCommand{\SN}{m}{
  \fp_compare:nNnTF { #1 }<{ 0 }{
    \fp_eval:n { 1-\sn{ \fp_abs:n {#1} } }
  }{
    \fp_eval:n { \sn{#1} }
  }
}

% claude.ai was used to help generate this macro
% Exact CDF of Student's t-distribution for positive-integer degrees of freedom
% Uses closed-form finite-sum formulas from Abramowitz & Stegun 26.7.3/26.7.8
% https://personal.math.ubc.ca/~cbm/aands/frameindex.htm

\fp_new:N \l__pt_s_fp     % sin \theta  =  t / sqrt(t^2+\nu)
\fp_new:N \l__pt_ct_fp    % cos^2\theta  =  \nu / (t^2+\nu)
\fp_new:N \l__pt_term_fp  % running series coefficient
\fp_new:N \l__pt_ctj_fp   % cos^{2j}\theta, accumulated
\fp_new:N \l__pt_sum_fp   % partial series sum
\fp_new:N \l__pt_val_fp   % result (always for t >= 0)

% Internal: compute CDF for t >= 0, store in \l__pt_val_fp
% #1 = t (non-negative), #2 = \nu (positive integer)
\cs_new_protected:Nn \__pt_cdf_pos:nn
  {
    \fp_set:Nn \l__pt_s_fp   { (#1) / sqrt( (#1)^2 + (#2) ) }
    \fp_set:Nn \l__pt_ct_fp  { (#2) / ( (#1)^2 + (#2) ) }
    \fp_set:Nn \l__pt_sum_fp  { 0 }
    \fp_set:Nn \l__pt_term_fp { 1 }  % P_0 = Q_0 = 1
    \fp_set:Nn \l__pt_ctj_fp  { 1 }  % cos^0 \theta = 1
    \int_if_odd:nTF { #2 }
      {% ---------- odd \nu = 2m+1, m = (\nu-1)/2 --------------------------------
       % F(t) = 1/2 + (1/\pi)·[arctan(t/\sqrt \nu) + sin\theta·cos\theta · \sum Q_j cos^{2j}\theta]
       % Q_j recurrence: Q_j = Q_{j-1} · 2j/(2j-1)
        \int_step_inline:nn { \int_eval:n { (#2 - 1) / 2 } }
          {
            \fp_add:Nn \l__pt_sum_fp { \l__pt_term_fp * \l__pt_ctj_fp }
            \fp_set:Nn \l__pt_ctj_fp  { \l__pt_ctj_fp  * \l__pt_ct_fp }
            \fp_set:Nn \l__pt_term_fp { \l__pt_term_fp * 2*##1 / (2*##1 + 1) }
          }
        \fp_set:Nn \l__pt_val_fp
          {
            0.5 + (1 / pi) * (
              atan( (#1) / sqrt(#2) )
              + \l__pt_s_fp * sqrt(\l__pt_ct_fp) * \l__pt_sum_fp
            )
          }
      }
      {% ---------- even \nu = 2m, m = \nu/2 -------------------------------------
       % F(t) = 1/2 + sin\theta/2 · \sum P_j cos^{2j}\theta
       % P_j recurrence: P_j = P_{j-1} · (2j-1)/(2j)
        \int_step_inline:nn { \int_eval:n { (#2) / 2 } }
          {
            \fp_add:Nn \l__pt_sum_fp { \l__pt_term_fp * \l__pt_ctj_fp }
            \fp_set:Nn \l__pt_ctj_fp  { \l__pt_ctj_fp  * \l__pt_ct_fp }
            \fp_set:Nn \l__pt_term_fp { \l__pt_term_fp * (2*##1 - 1) / (2*##1) }
          }
        \fp_set:Nn \l__pt_val_fp { 0.5 + \l__pt_s_fp / 2 * \l__pt_sum_fp }
      }
  }

% Internal: compute CDF for any t, store in \l__pt_val_fp
\cs_new_protected:Nn \__pt_cdf:nn
  {
  \__pt_cdf_pos:nn { \fp_eval:n { abs(#1) } } {#2}
    \fp_compare:nNnTF { \fp_eval:n {#1} } < { 0 }
      { \fp_set:Nn \l__pt_val_fp { 1 - \l__pt_val_fp } }
      { }
  }

\fp_new:N \l__stat_tdist_prob
\cs_new_protected:Nn \__stat_compute_ptdist: {
  \fp_set:Nn \l__stat_tdist_prob { 1 }
  \bool_if:NT \l__stat_kv_upper_explicit_bool {
    \__pt_cdf:nn { \fp_eval:n {
      (\l__stat_kv_upper_fp - \l__stat_kv_mean_fp) / \l__stat_kv_stdev_fp }}{
      \l__stat_kv_df_int }
    \fp_set_eq:NN \l__stat_tdist_prob \l__pt_val_fp
  }
  \bool_if:NT \l__stat_kv_lower_explicit_bool {
    \__pt_cdf:nn { \fp_eval:n {
      (\l__stat_kv_lower_fp - \l__stat_kv_mean_fp) / \l__stat_kv_stdev_fp }}{
      \l__stat_kv_df_int }
    \fp_sub:Nn \l__stat_tdist_prob { \fp_eval:n \l__pt_val_fp }
  }
}

%% ============================================================
%% Quantile function (inverse CDF) for the t-distribution
%% Requires \__pt_cdf:nn from the CDF implementation above.
%% ============================================================
%% code generated using Claude.ai
\fp_new:N \l__qt_lo_fp
\fp_new:N \l__qt_hi_fp
\fp_new:N \l__qt_mid_fp

%% Internal bisection engine.
%% 54 steps: width 100 / 2^54 = 5.5e-15 (sub-ULP precision).
\cs_new_protected:Nn \__qt_bisect:nn
  {% #1 = p (probability), #2 = nu (positive integer df)
    \fp_set:Nn \l__qt_lo_fp { -50 }
    \fp_set:Nn \l__qt_hi_fp {  50 }
    \int_step_inline:nn { 54 }
      {
        \fp_set:Nn \l__qt_mid_fp { ( \l__qt_lo_fp + \l__qt_hi_fp ) / 2 }
        \__pt_cdf:nn { \l__qt_mid_fp } {#2}
        \fp_compare:nNnTF { \l__pt_val_fp } < { #1 }
          { \fp_set_eq:NN \l__qt_lo_fp \l__qt_mid_fp }
          { \fp_set_eq:NN \l__qt_hi_fp \l__qt_mid_fp }
      }
    \fp_set:Nn \l__qt_mid_fp { ( \l__qt_lo_fp + \l__qt_hi_fp ) / 2 }
  }

%%   Standard quantile: returns t such that P(T_{df} <= t) = p.
%\qt{keys: pvalue, df, var, format}
\NewDocumentCommand \qt { m }{
  \group_begin:
  \bool_set_false:N \l__stat_kv_lower_explicit_bool
  \bool_set_false:N \l__stat_kv_upper_explicit_bool
  \bool_set_false:N \l__stat_kv_var_explicit_bool
  \keys_set:nn { stat }{ #1 }
  \__qt_bisect:nn { \fp_eval:n { \l__stat_kv_pvalue_fp } } { \l__stat_kv_df_int }
  \bool_if:NTF \l__stat_kv_var_explicit_bool {
    \cs_gset:cpx { \l__stat_kv_store_csname }{ \fp_to_decimal:N \l__qt_mid_fp }
  }{
    \exp_args:NnV \fp_format:nn { \l__qt_mid_fp } \l__stat_fmt_tl
  }
  \group_end:
}

%%   Returns the POSITIVE t* such that P(|T_{df}| > t*) = alpha,
