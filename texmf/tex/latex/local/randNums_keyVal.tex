\int_new:N    \l__dp_lower_int   % decimal places in lower bound string
\int_new:N    \l__dp_upper_int   % decimal places in upper bound string
\fp_new:N     \l__dp_scale_fp    % = 1/step %%% = 10^max(dp_lower, dp_upper)
\seq_new:N    \l__dp_seq         % scratch for splitting on "."
\tl_new:N     \l__dp_frac_tl     % scratch for the fractional part
%% count num decimal places on bounds
\cs_new_protected:Npn \__randints_count_dp:NN #1 #2
{
  \tl_set:Nx \l__dp_frac_tl { \tl_use:N #2 }
  % Split on ".": one item = integer, two items = has decimal part
  \seq_set_split:NnV \l__dp_seq { . } \l__dp_frac_tl
  \int_compare:nNnTF { \seq_count:N \l__dp_seq } = { 1 }
  { \int_set:Nn #1 { 0 } }
  {
    \tl_set:Nx \l__dp_frac_tl { \seq_item:Nn \l__dp_seq { 2 } }
    \int_set:Nn #1 { \tl_count:N \l__dp_frac_tl }
  }
}

\bool_new:N   \l__pick_found_bool           % track if rand num in skip
\bool_new:N   \l__exam_randNums_uniq_bool   % [uniq] or star
\int_new:N    \l__exam_randNums_min_int     % sorted min/max
\int_new:N    \l__exam_randNums_max_int     % sorted min/max
\int_new:N    \l__pool_size_int             % stores size of pool of ints
\int_new:N    \l__sort_idx_int              % index for sorting values
\seq_new:N    \l__sort_seq                  % sequence to hold values for sorting
\tl_new:N     \l__my_pick_tl                % scratch for randomly chosen value

%keyval variables
\bool_new:N   \l__exam_randNums_kv_nonzero_bool
\bool_new:N   \l__exam_randNums_kv_uniq_bool
\bool_new:N   \l__exam_randNums_kv_sort_bool
\bool_new:N   \l__exam_randNums_kv_step_explicit_bool
\clist_new:N  \l__exam_randNums_kv_skip_clist
\fp_new:N     \l__exam_randNums_kv_min_fp
\fp_new:N     \l__exam_randNums_kv_max_fp
\fp_new:N     \l__exam_randNums_kv_mult_fp
\fp_new:N     \l__exam_randNums_kv_step_fp

\keys_define:nn { exam / randNums }
  {
    nz            .bool_set:N  = \l__exam_randNums_kv_nonzero_bool,
    nz            .initial:n   = false,
    nz            .default:n   = true,
    %
    sort          .bool_set:N  = \l__exam_randNums_kv_sort_bool,
    sort          .initial:n   = false,
    sort          .default:n   = true,
    %
    uniq          .bool_set:N  = \l__exam_randNums_kv_uniq_bool,
    uniq          .initial:n   = false,
    uniq          .default:n   = true,
    %
    skip          .clist_set:N = \l__exam_randNums_kv_skip_clist,
    %
    min           .code:n      = {
                    \fp_set:Nn  \l__exam_randNums_kv_min_fp { #1 }
                    \tl_set:Nn  \l__exam_randNums_kv_min_tl { #1 } },
    min           .initial:n   = 1,
    %
    max           .code:n      = {
                    \fp_set:Nn  \l__exam_randNums_kv_max_fp { #1 }
                    \tl_set:Nn  \l__exam_randNums_kv_max_tl { #1 } },
    max           .initial:n   = 5,
    %
    mult          .fp_set:N    = \l__exam_randNums_kv_mult_fp,
    mult          .initial:n   = 1,
    %
    step          .code:n      = {
                    \fp_set:Nn  \l__exam_randNums_kv_step_fp { #1 }
                    \bool_set_true:N \l__exam_randNums_kv_step_explicit_bool },
    step          .initial:n   = 1,
  }

\msg_new:nnn { randNums } { too-many }
  { \randNums*:~only~#1~value(s)~available~in~[#3]~but~#2~requested. }

% \randNums * [keys] {csname1,csname2,...}
% #1 - starred variant produces unique values
% #2 - optional args: (default)
%          nz = (false) generate nonzero values from [-max,-min]U[min,max]
%        sort = (false) sort random values so that csname1<csname2<...
%        uniq = (false) produce unique values
%        skip = () clist of values to skip
%         min = (1) minimum value of range
%         max = (5) maximum value of range
%        mult = (1) value to multiply by after drawing (e.g. random multiples of 5)
% #3 - clist of names; defines \name globally to the drawn value
\NewDocumentCommand \randNums { s O{} m }{
  \group_begin: %grouping used to keep key changes local
  %TODO: why does this key need to be explicitly reset?
  \bool_set_false:N \l__exam_randNums_kv_step_explicit_bool
  \keys_set:nn { exam / randNums }{ #2 }

  % Star is equivalent to uniq
  \bool_if:nT {#1} { \bool_set_true:N \l__exam_randNums_kv_uniq_bool }

\typeout{!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!}
\bool_if:NTF \l__exam_randNums_kv_nonzero_bool {\typeout{DEBUG~nz:~true}}{\typeout{DEBUG~nz:~false}}
\bool_if:NTF \l__exam_randNums_kv_sort_bool {\typeout{DEBUG~sort:~true}}{\typeout{DEBUG~sort:~false}}
\bool_if:NTF \l__exam_randNums_kv_uniq_bool {\typeout{DEBUG~uniq:~true}}{\typeout{DEBUG~uniq:~false}}
\bool_if:NTF \l__exam_randNums_kv_step_explicit_bool {\typeout{DEBUG~step:~true}}{\typeout{DEBUG~step:~false}}
\clist_map_inline:nn {\l__exam_randNums_kv_skip_clist}{\typeout{DEBUG~skip: ##1 }}
\typeout{DEBUG~min/max:~\fp_eval:n {\l__exam_randNums_kv_min_tl},~\fp_eval:n {\l__exam_randNums_kv_max_tl}}
\typeout{DEBUG~mult:~\fp_eval:n {\l__exam_randNums_kv_mult_fp}}
\typeout{DEBUG~step:~\fp_eval:n {\l__exam_randNums_kv_step_fp}}

  % Set uniq_bool based on keyval pair
  \bool_set_false:N \l__exam_randNums_uniq_bool
  \bool_if:NT \l__exam_randNums_kv_uniq_bool { \bool_set_true:N \l__exam_randNums_uniq_bool }

  % % Detect decimal places in each bound's string form
  \__randints_count_dp:NN \l__dp_lower_int { \l__exam_randNums_kv_min_tl }
  \__randints_count_dp:NN \l__dp_upper_int { \l__exam_randNums_kv_max_tl }


  % Scale factor = 10^(max decimal places); 1 when both bounds are integers
  \fp_set:Nn \l__dp_scale_fp
  { \fp_eval:n { 10 ^ \int_max:nn { \l__dp_lower_int } { \l__dp_upper_int } } }

  %manually override scaling
  \bool_if:NT \l__exam_randNums_kv_step_explicit_bool {
    \fp_set:Nn \l__dp_scale_fp { \fp_eval:n { 1 / \l__exam_randNums_kv_step_fp }}
  }

% \typeout{DEBUG~dp_lower/upper_int: \int_eval:n {\l__dp_lower_int},~\int_eval:n {\l__dp_upper_int}}
\typeout{DEBUG~dp_scale_fp: \fp_eval:n {\l__dp_scale_fp}}

  %TODO fix this: prints garbage when step produces a non-integer value
  % set the upper and lower bounds for the random integers
  \int_set:Nn \l__exam_randNums_min_int {
    \fp_eval:n{ min(
      \l__exam_randNums_kv_min_fp,
      \l__exam_randNums_kv_max_fp )
      * \l__dp_scale_fp  }
  }
  \int_set:Nn \l__exam_randNums_max_int {
    \fp_eval:n{ max(
      \l__exam_randNums_kv_min_fp,
      \l__exam_randNums_kv_max_fp )
      * \l__dp_scale_fp }
  }

\typeout{DEBUG~scaled~min/max:~\int_eval:n { \l__exam_randNums_min_int },~\int_eval:n { \l__exam_randNums_max_int }}

  %build a sequence of "used values"
  \seq_clear_new:N \l__used_seq
  \clist_map_inline:Nn \l__exam_randNums_kv_skip_clist {
    \bool_if:NTF { \l__exam_randNums_kv_nonzero_bool }{
      % nz pool: ±[lower..upper]*multiplier, so check |value| is in range
      \fp_compare:nNnTF { abs(##1) } < { \l__exam_randNums_min_int * \l__exam_randNums_kv_mult_fp}{
        % out of range low - skip
        }{
          \fp_compare:nNnTF { abs(##1) } > { \l__exam_randNums_max_int * \l__exam_randNums_kv_mult_fp }{
            % out of range high - skip
          }{
            \seq_put_right:Nx \l__used_seq { \fp_eval:n { ##1 } }
          }
      }
    }{
      % normal pool: check lower*multiplier <= value <= upper*multiplier
      \fp_compare:nNnTF { ##1 } < {
        \l__exam_randNums_min_int /
        \l__dp_scale_fp * \l__exam_randNums_kv_mult_fp }
        { } % out of range - skip
        {
          \fp_compare:nNnTF { ##1 } > {
            \l__exam_randNums_max_int / \l__dp_scale_fp *
            \l__exam_randNums_kv_mult_fp }{
            % out of range - skip
          }{
            \seq_put_right:Nx \l__used_seq { \fp_eval:n { ##1 } }
          }
        }
    }
  }

  % overflow check (only meaningful without replacement)
  \bool_if:NTF { \l__exam_randNums_uniq_bool }{
    \bool_if:NTF { \l__exam_randNums_kv_nonzero_bool }
      { \int_set:Nn \l__pool_size_int { 2 * ( \l__exam_randNums_max_int - \l__exam_randNums_min_int + 1 ) } }
      { \int_set:Nn \l__pool_size_int {       \l__exam_randNums_max_int - \l__exam_randNums_min_int + 1   } }
    % Subtract the number of values being skipped
    \int_sub:Nn \l__pool_size_int { \seq_count:N \l__used_seq }
    % Compare against the number of values requested
    \int_compare:nNnT { \clist_count:n { #3 } } > { \l__pool_size_int }{
      \msg_fatal:nnxxx { randNums } { too-many }
        { \int_use:N \l__pool_size_int }
        { \clist_count:n { #3 } }
        { \bool_if:NTF { \l__exam_randNums_kv_nonzero_bool } {nz~}
          {} \int_eval:n { \l__exam_randNums_min_int }
          .. \int_eval:n { \l__exam_randNums_max_int }}%
    }
  }{
  }

  %generate random integers
  \clist_map_inline:nn {#3}{
    \bool_set_false:N \l__pick_found_bool
    \bool_until_do:Nn \l__pick_found_bool{
      \bool_if:NTF { \l__exam_randNums_kv_nonzero_bool }{
        \tl_set:Nx \l__my_pick_tl{
          \fp_eval:n{
            \int_rand:nn {1}{2} == 1
              ? \int_rand:nn
                {\l__exam_randNums_min_int}
                {\l__exam_randNums_max_int}
                / \l__dp_scale_fp * \l__exam_randNums_kv_mult_fp
              : -\int_rand:nn
                {\l__exam_randNums_min_int}
                {\l__exam_randNums_max_int}
                / \l__dp_scale_fp * \l__exam_randNums_kv_mult_fp
            }
          }
        }{
          \tl_set:Nx \l__my_pick_tl
            { \fp_eval:n { \int_rand:nn
                {\l__exam_randNums_min_int}
                {\l__exam_randNums_max_int}
                / \l__dp_scale_fp * \l__exam_randNums_kv_mult_fp  } }
        }
      % Accept the pick only if it's not in the exclusion list
      \seq_if_in:NVF \l__used_seq \l__my_pick_tl { \bool_set_true:N \l__pick_found_bool }
    }

    %Add value generated to used sequence if done w/o replacement
    \bool_if:NTF { \l__exam_randNums_uniq_bool }{ \seq_put_right:NV \l__used_seq \l__my_pick_tl }{}
    %Globally store value generated
    \cs_gset:cpx {##1}{ \tl_use:N \l__my_pick_tl }
  }

  %sort values
  \bool_if:NT \l__exam_randNums_kv_sort_bool {
    \seq_clear:N \l__sort_seq
    \clist_map_inline:nn {#3}
      { \seq_put_right:Nx \l__sort_seq { \use:c { ##1 } } }   % \use:c dereferences
    \seq_sort:Nn \l__sort_seq {
      \fp_compare:nNnTF { ##1 } > { ##2 }
        { \sort_return_swapped: }
        { \sort_return_same: }
    }
    \int_zero:N \l__sort_idx_int
    \clist_map_inline:nn {#3} {
      \int_incr:N \l__sort_idx_int
      \cs_gset:cpx {##1} { \seq_item:Nn \l__sort_seq { \l__sort_idx_int } }
    }
  }
  \group_end:
}
