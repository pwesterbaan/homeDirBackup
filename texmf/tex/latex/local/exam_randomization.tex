  -0.5*x^2 -1.26551223
  +1.00002368*h    +0.37409196*h^2
  +0.09678418*h^3 -0.18628806*h^4
  +0.27886807*h^5 -1.13520398*h^6
  +1.48851587*h^7 -0.82215223*h^8
  +0.17087277*h^9);

\int_new:N \l__lower_bnd
\int_new:N \l__upper_bnd
\int_new:N \l__pool_size_int
\tl_new:N  \l__my_pick_tl
\bool_new:N \l__pick_found_bool

\msg_new:nnn { randInts } { too-many }
  { \randInts*:~only~#1~value(s)~available~in~[#3]~but~#2~requested. }

\NewDocumentCommand \randInts { s o D<>{\randIntLower} D(){\randIntUpper} m O{1} D<>{} }{
  % \randInts[nz?]<min>(max){name1,name2,...}[t]
  % - [nz]: produce pm(magnitude), where magnitude in [min,max]
  % - <min> and (max) are *separate* optional delimited args; defaults are 1 and 5
  % - clist of names; defines \name globally to the drawn integer
  % - optional randInt multiplier t (defaults to 1)
  % - optional clist of values to skip

  %set the upper and lower bounds for the random integers
  \int_set:Nn \l__lower_bnd { \int_min:nn {#3}{#4} }
  \int_set:Nn \l__upper_bnd { \int_max:nn {#3}{#4} }

  %build a sequence of "used integers"
  %starting with the values to avoid in #7
  \seq_clear_new:N \l__used_seq
  \clist_map_inline:nn {#7}{
    \tl_if_eq:nnTF {#2}{nz}{
      % nz pool: ±[lower..upper], so check |value| is in range
      \int_compare:nNnTF { \int_abs:n {##1} } < { \l__lower_bnd }{
        % out of range low - skip
        }{
          \int_compare:nNnTF { \int_abs:n {##1} } > { \l__upper_bnd }{
            % out of range high - skip
          }{
            \seq_put_right:Nn \l__used_seq {##1}
          }
      }
    }{
      % normal pool: check lower ≤ value ≤ upper
      \int_compare:nNnTF
        { ##1 } < { \l__lower_bnd }
        { } % out of range - skip
        {
          \int_compare:nNnTF { ##1 } > { \l__upper_bnd }{
            % out of range - skip
          }{
            \seq_put_right:Nn \l__used_seq {##1}
          }
        }
    }
  }

  % overflow check (only meaningful without replacement)
  \IfBooleanT {#1}{
    \tl_if_eq:nnTF {#2}{nz}
      { \int_set:Nn \l__pool_size_int { 2 * ( \l__upper_bnd - \l__lower_bnd + 1 ) } }
      { \int_set:Nn \l__pool_size_int {       \l__upper_bnd - \l__lower_bnd + 1   } }
    % Subtract the number of values being skipped
    \int_sub:Nn \l__pool_size_int { \seq_count:N \l__used_seq }
    % Compare against the number of values requested
    \int_compare:nNnT { \clist_count:n {#5} } > { \l__pool_size_int }{
      \msg_fatal:nnxxx { randInts } { too-many }
        { \int_use:N \l__pool_size_int }
        { \clist_count:n {#5} }
        { \tl_if_eq:nnTF {#2}{nz} {nz~} {} #3 .. #4 }
    }
  }

  %generate random integers
  \clist_map_inline:nn {#5}{
    \bool_set_false:N \l__pick_found_bool
    \bool_until_do:Nn \l__pick_found_bool{
      \tl_if_eq:nnTF {#2}{nz}{
        \tl_set:Nx \l__my_pick_tl{
          \fp_eval:n{
            \int_rand:nn {1}{2} == 1
              ? \int_rand:nn {\l__lower_bnd}{\l__upper_bnd} * #6
              : -\int_rand:nn {\l__lower_bnd}{\l__upper_bnd} * #6
            }
          }
        }{
          \tl_set:Nx \l__my_pick_tl
            { \fp_eval:n { \int_rand:nn {\l__lower_bnd}{\l__upper_bnd} * #6 } }
        }
      % Accept the pick only if it's not in the exclusion list
      \seq_if_in:NVF \l__used_seq \l__my_pick_tl { \bool_set_true:N \l__pick_found_bool }
    }

    %Add value generated to used sequence if done w/o replacement
    \IfBooleanT {#1}{ \seq_put_right:NV \l__used_seq \l__my_pick_tl }
    %Globally store value generated
    \cs_gset:cpx {##1}{ \tl_use:N \l__my_pick_tl }
  }
}

\bool_new:N \l_isprime_bool
\int_new:N \l_trial_int

\NewDocumentCommand \IfPrime { m m m } {
  % #1 = number to test, #2 = true code, #3 = false code
  \bool_set_true:N \l_isprime_bool

  \int_compare:nNnT { #1 } < { 2 } { \bool_set_false:N \l_isprime_bool }

  \int_if_even:nT { #1 } {
    \int_compare:nNnT { #1 } > { 2 } { \bool_set_false:N \l_isprime_bool }
  }

  % Trial division by odd numbers from 3 up to floor(sqrt(#1))
  \int_set:Nn \l_trial_int { 3 }
  \bool_while_do:nn
    {
      \l_isprime_bool
      &&
      \int_compare_p:n { \l_trial_int * \l_trial_int <= #1 }
    }
    {
      \int_compare:nNnT { \int_mod:nn{#1}{\l_trial_int} } = { 0 }
        { \bool_set_false:N \l_isprime_bool }
      \int_add:Nn \l_trial_int { 2 }
    }

  \bool_if:NTF \l_isprime_bool { #2 } { #3 }
}

\int_new:N \l_candidate_int

\NewDocumentCommand \RandPrime { m m m } {
  % #1 control sequence, #2 = lower bound, #3 = upper bound
  % Result stored in \l_candidate_int (and printed)
  \int_if_exist:NF #1 { \int_new:N #1 }
  \bool_set_false:N \l_isprime_bool
  \bool_until_do:Nn \l_isprime_bool {
    \int_set:Nn #1 { \int_rand:nn{#2}{#3} }
    \IfPrime{ #1 }
      { \bool_set_true:N \l_isprime_bool }
      {}
  }
}

% math125 exam02: computing powers mod m
% Returns true if b has order exactly 4 mod m
% i.e. b^2 != 1 (mod m)  AND  b^4 == 1 (mod m)
\NewDocumentCommand \IfOrderFour { m m m m } {
  % #1=base, #2=modulus, #3=true code, #4=false code
  \int_set:Nn \l_tmpa_int { \int_mod:nn { #1 * #1 } { #2 } }           % b^2 mod m
  \int_set:Nn \l_tmpb_int { \int_mod:nn { \l_tmpa_int * \l_tmpa_int } { #2 } }  % b^4 mod m
  \int_compare:nNnTF { \l_tmpb_int } = { 1 }
    {
      \int_compare:nNnTF { \l_tmpa_int } = { 1 }
        { #4 }  % b^2 == 1, order is 1 or 2, not 4
        { #3 }  % b^4 == 1 but b^2 != 1: order is exactly 4 ✓
    }
    { #4 }  % b^4 != 1
}

% Pick random base with order 4 mod m
\NewDocumentCommand \RandOrderFourBase { m m m m } {
  % #1 = control seq for base, #2 = modulus
  % #3 = lower bound for base search, #4 = upper bound
  \int_if_exist:NF #1 { \int_new:N #1 }
  \bool_set_false:N \l_isprime_bool  % reuse as "found" flag
  \bool_until_do:Nn \l_isprime_bool {
    \int_set:Nn #1 { \int_rand:nn { #3 } { #4 } }
    \IfOrderFour { \int_use:N #1 } { #2 }
      { \bool_set_true:N \l_isprime_bool }
      {}
  }
}

\NewDocumentCommand \IfOrderThree { m m m m } {
  % #1=base, #2=modulus, #3=true, #4=false
  % Check: b^3 ≡ 1 (mod m)  AND  b ≢ 1 (mod m)
  \int_set:Nn \l_tmpa_int { \int_mod:nn { #1 * #1 }           { #2 } }  % b^2
  \int_set:Nn \l_tmpb_int { \int_mod:nn { \l_tmpa_int * #1 }  { #2 } }  % b^3
  \int_compare:nNnTF { \l_tmpb_int } = { 1 }
    {
      \int_compare:nNnTF { #1 } = { 1 }
        { #4 }   % b = 1, trivial, reject
        { #3 }   % b^3 ≡ 1 and b ≠ 1: order exactly 3 ✓
    }
    { #4 }
}

\NewDocumentCommand \RandOrderThreeBase { m m m m } {
  % #1=control seq, #2=modulus, #3=lower, #4=upper
  \int_if_exist:NF #1 { \int_new:N #1 }
  \bool_set_false:N \l_isprime_bool
  \bool_until_do:Nn \l_isprime_bool {
    \int_set:Nn #1 { \int_rand:nn { #3 } { #4 } }
    \IfOrderThree { \int_use:N #1 } { #2 }
      { \bool_set_true:N \l_isprime_bool }
      {}
  }
}

\int_new:N \l_tmpc_int
\NewDocumentCommand \IfGoodAlternating { m m m m } {
  % #1=base, #2=modulus, #3=true, #4=false
  % Requires: b^3 ≡ 1 (mod m), b ≢ 0,1 (mod m), b ≢ m-1 (mod m)
  \int_set:Nn \l_tmpa_int { \int_mod:nn { #1 } { #2 } }           % b mod m
  \int_set:Nn \l_tmpb_int { \int_mod:nn { \l_tmpa_int * \l_tmpa_int } { #2 } }  % b^2
  \int_set:Nn \l_tmpc_int { \int_mod:nn { \l_tmpb_int * \l_tmpa_int } { #2 } }  % b^3

  % Reject degenerate residues
  \bool_set_false:N \l_isprime_bool
  \int_compare:nNnT { \l_tmpa_int } > { 1 }  % b mod m > 1
    {
      \int_compare:nNnT { \l_tmpa_int } < { \int_eval:n{#2 - 1} }  % b mod m < m-1
        {
          \int_compare:nNnT { \l_tmpb_int } > { 1 }  % r = b^2 mod m > 1
            {
              \int_compare:nNnT { \l_tmpc_int } = { 1 }  % b^3 ≡ 1
                { \bool_set_true:N \l_isprime_bool }
            }
        }
    }
  \bool_if:NTF \l_isprime_bool { #3 } { #4 }
}

\NewDocumentCommand \RandAlternatingBase { m m m m } {
  % #1=control seq for base, #2=modulus, #3=lower, #4=upper
  \int_if_exist:NF #1 { \int_new:N #1 }
  \bool_set_false:N \l_isprime_bool
  \bool_until_do:Nn \l_isprime_bool {
    \int_set:Nn #1 { \int_rand:nn { #3 } { #4 } }
    \IfGoodAlternating { \int_use:N #1 } { #2 }
      { \bool_set_true:N \l_isprime_bool }
      {}
  }
}

\NewDocumentCommand \RandPrimeModThree { m m m } {
  % #1=control seq, #2=lower, #3=upper
  % Finds a random prime p in [lower,upper] with p ≡ 1 (mod 3)
  \int_if_exist:NF #1 { \int_new:N #1 }
  \bool_set_false:N \l_isprime_bool
  \bool_until_do:Nn \l_isprime_bool {
    \RandPrime #1 { #2 } { #3 }
    \int_compare:nNnT { \int_mod:nn { \int_use:N #1 } { 3 } } = { 1 }
      { \bool_set_true:N \l_isprime_bool }
  }
}

\cs_new_protected:Npn \exam_print_coeff:nnn #1 #2 #3
  {
    % Evaluate coefficient once via pgf
    % TODO Replace pgfmathparse with xparse
    % \pgfmathparse{#3}
    % \tl_set:Nx \l_tmpa_tl { \pgfmathresult }  % textual form for pgf printer
    \tl_set:Nx \l_tmpa_tl { \fp_eval:n { #3 } } % textual form for pgf printer
    \fp_set:Nn \l_tmpa_fp { \l_tmpa_tl }        % numeric form for comparisons

    % Star: default signed; star => suppress leading '+'
    \bool_set_true:N \l_exam_print_sign_bool
    \bool_if:nT {#1} { \bool_set_false:N \l_exam_print_sign_bool }

    % Optional keyword: [signs] (case-insensitive; spaces ignored if present)
    \bool_set_false:N \l_exam_signonly_units_bool
    \str_if_eq:eeTF
      { \tl_lower_case:n { \tl_trim_spaces:n {#2} } }
      { signs }
      { \bool_set_true:N \l_exam_signonly_units_bool }
      { }

    % 0 -> print nothing
    \fp_compare:nTF { \l_tmpa_fp == 0 }
      { \ensuremath{} }
      {
        % -1
        \fp_compare:nTF { \l_tmpa_fp == -1 }
          {
            \bool_if:NTF \l_exam_signonly_units_bool {
              \ensuremath{-} } { \ensuremath{-1} }
          }
          {
            % +1
            \fp_compare:nTF { \l_tmpa_fp == 1 }
              {
                \bool_if:NTF \l_exam_signonly_units_bool
                  {  \bool_if:NTF \l_exam_print_sign_bool { % '+' or empty
                    \ensuremath{+} } { \ensuremath{} }
                  }{ \bool_if:NTF \l_exam_print_sign_bool {
                    \ensuremath{+1} } { \ensuremath{1} } }
              }{
                % general numeric case
                \group_begin:
                  %TODO \fp_format:nn or \fp_to_decimal:c
                  %TODO use \fprintsigned
                  \bool_if:NTF \l_exam_print_sign_bool
                    {
                    \ensuremath{\pgfmathprintnumber[1000~sep={,},print~sign,fixed]{\l_tmpa_tl}}
                    }
                    % { \ensuremath{\pgfmathprintnumber{\l_tmpa_tl}} }
                    { \ensuremath{ \num[group-separator={,}]{
                      \fp_to_decimal:n { round(\l_tmpa_tl,
                      \numDecPlaces) } }
                      }
                    }
                \group_end:
              }
          }
      }
  }

    \tl_if_eq:nnTF {#1}{\geq}{\leq}{

% Pad a *binary string* to the next multiple of 4 (nibble)
\cs_new:Npn \bin_pad_to_nibble:n #1
  {
    % IMPORTANT: expand the input (#1) so macros like \pgfmathresult are resolved
    \tl_set:Nx \l__bin_tl { #1 }

    % Sanitize: trim edges and remove internal spaces
    \tl_trim_spaces:N \l__bin_tl
    \tl_remove_all:Nn \l__bin_tl { \c_space_token }

    % Count and compute padding
    \int_set:Nn \l__bin_len_int { \tl_count:N \l__bin_tl }
    \int_set:Nn \l__bin_mod_int { \int_mod:nn { \l__bin_len_int } { 4 } }

    \int_compare:nNnT { \l__bin_mod_int } > { 0 }
      {
        \int_set:Nn \l__bin_pad_int { \int_eval:n { 4 - \l__bin_mod_int } }
        \int_step_inline:nn { \l__bin_pad_int }
          { \tl_put_left:Nn \l__bin_tl { 0 } }
      }

    % Output
    \tl_use:N \l__bin_tl
  }

% Set random seed based on year, semester, and version
% e.g. VD in spring 2026 corresponds to 1326
% year:           26 : \int_mod:nn { \year }{ 100 }
% exam version   300 : 100 * ( \verNum - 1 )
% semester      1000 : \int_eval:n { \month * 100 + \day

\int_new:N \l__exam_year_two_int
\int_new:N \l__exam_md_int
\int_new:N \l__exam_sem_offset_int
\int_new:N \l__exam_seed_int

% two-digit year
\int_set:Nn \l__exam_year_two_int { \int_mod:nn { \year } { 100 } }

% month-day as MMDD integer (e.g. May 14 -> 514)
\int_set:Nn \l__exam_md_int { \int_eval:n { \month * 100 + \day } }

% semester-based offset: before May 15 -> 1000, before Jun 25 -> 2000,
% before Aug 15 -> 3000, otherwise 4000
\int_set:Nn \l__exam_sem_offset_int { 4000 }
\int_compare:nNnTF { \l__exam_md_int } < { 515 }
  { \int_set:Nn \l__exam_sem_offset_int { 1000 } }          %spring
  {
    \int_compare:nNnTF { \l__exam_md_int } < { 625 }
      { \int_set:Nn \l__exam_sem_offset_int { 2000 } }      %summer I
      {
        \int_compare:nNnTF { \l__exam_md_int } < { 815 }
          { \int_set:Nn \l__exam_sem_offset_int { 3000 } }  %summer II
          { \int_set:Nn \l__exam_sem_offset_int { 4000 } }  %fall
      }
  }

% base seed = semester offset + two-digit year
\int_set:Nn \l__exam_seed_int { \l__exam_sem_offset_int + \l__exam_year_two_int }

% if a `\version` macro exists, convert to numeric and offset by 100*(ver-1)
\cs_if_exist:NTF \version
  {
    \letterToNum{\verNum}{\version}
    \int_set:Nn \l__exam_seed_int { \int_eval:n { \l__exam_seed_int + 100 * ( \verNum - 1 ) } }
  }
  { }

% Set global random seed (query with \sys_rand_seed:
\sys_gset_rand_seed:n { \int_use:N \l__exam_seed_int }
