% Key family for \printCoeff
%   signsOnly     - for ±1, print just the sign
%   zero          - force-print zero as "0"
%   noplus        - suppress leading '+' (alternative to star)
%   format=<fmt>  - fp_format string, e.g. format=0.2f
%   numDec        - number of decimal places to round to
\keys_define:nn { exam / printCoeff }
  {
    %TODO Double check which keys need defaults, inits, etc.
    signs-only    .bool_set:N  =  \l_exam_signonly_units_bool ,
    signs-only    .initial:n   =  true ,
    signs-only    .default:n   =  true , %set when called without setting
    %
    zero          .bool_set:N  =  \l_exam_print_zero_bool ,
    zero          .initial:n   =  false ,
    zero          .default:n   =  true , %set when called without setting
    %
    noplus        .bool_set:N  =  \l__printcoeff_noplus_bool ,
    noplus        .initial:n   =  false ,
    noplus        .default:n   =  true , %set when called without setting
    %
    format        .tl_set:N    =  \l__printcoeff_fmt_tl ,
    %
    numDecPlaces  .int_set:N   =  \l__numDecPlaces,
    numDecPlaces  .initial:n   =  4,
  }

% \printCoeff * [keys] {<coeff>}
%   star:           suppress leading '+' (equivalent to noplus key)
%   [signs-only]:   for ±1, print just the sign
%   [zero]:         force-print zero as "0"
%   [noplus]:       suppress leading '+' without the star
%   [format=<fmt>]: fp_format string (default: empty)
\NewDocumentCommand \printCoeff { s O{} m }
  { \exam_print_coeff:nnn {#1} {#2} {#3} }

\cs_new_protected:Npn \exam_print_coeff:nnn #1 #2 #3
  {
    \group_begin:
%%%%%    % Reset all keys to per-invocation defaults, then apply user's keys
%%%%%%    \keys_set:nn { exam / printCoeff }
%%%%%%      { signs-only, zeros, noplus, format={} }
      \keys_set:nn { exam / printCoeff } { #2 }

      % Star is equivalent to noplus
      \bool_if:nT {#1} { \bool_set_true:N \l__printcoeff_noplus_bool }

      % Derive print_sign from noplus
      \bool_set_true:N \l_exam_print_sign_bool
      \bool_if:NT \l__printcoeff_noplus_bool
        { \bool_set_false:N \l_exam_print_sign_bool }

      \tl_set:Nx \l_tmpa_tl { \fp_eval:n { #3 } } % textual form for printing
      \fp_set:Nn \l_tmpa_fp { \l_tmpa_tl }        % numeric form for comparisons

      %TODO consider case where num=-1,0,1 with format nonempty

      % 0
      \fp_compare:nTF { \l_tmpa_fp == 0 }
        {
          \bool_if:NTF \l_exam_print_zero_bool
            {
              \bool_if:NTF \l_exam_print_sign_bool
                { \ensuremath{+0} }
                { \ensuremath{0}  }
            }
            { \ensuremath{} }
        }
        {
          % -1
          \fp_compare:nTF { \l_tmpa_fp == -1 }
            {
              \bool_if:NTF \l_exam_signonly_units_bool
                { \ensuremath{-}  }
                { \ensuremath{-1} }
            }
            {
              % +1
              \fp_compare:nTF { \l_tmpa_fp == 1 }
                {
                  \bool_if:NTF \l_exam_signonly_units_bool
                    { \bool_if:NTF \l_exam_print_sign_bool
                        { \ensuremath{+} }
                        { \ensuremath{}  }
                    }
                    { \bool_if:NTF \l_exam_print_sign_bool
                        { \ensuremath{+1} }
                        { \ensuremath{1}  }
                    }
                }
                {
                  % general numeric case
                  \group_begin:
                    \ensuremath{
                      \bool_if:NTF \l_exam_print_sign_bool
                        { \num[group-separator={,}, group-digits=integer, print-implicit-plus=true]{
                          \exp_args:NnV \fp_format:nn
                            { round(\l_tmpa_tl, \l__numDecPlaces) } \l__printcoeff_fmt_tl } }
                        { \num[group-separator={,}, group-digits=integer]{
                          \exp_args:NnV \fp_format:nn
                            { round(\l_tmpa_tl, \l__numDecPlaces) } \l__printcoeff_fmt_tl } }
                    }
                  \group_end:
                }
            }
        }
    \group_end:
    }
\ExplSyntaxOff%
